rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /fotos/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['autor','img','ts'])
        && request.resource.data.autor is string
        && request.resource.data.autor.size() > 0
        && request.resource.data.autor.size() <= 60
        && request.resource.data.img is string
        && request.resource.data.img.size() < 950000
        && request.resource.data.img.matches('data:image/jpeg;base64,.*')
        && request.resource.data.ts is number;
      allow update, delete: if false;
    }

    match /mensajes/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['nombre','texto','ts'])
        && request.resource.data.nombre is string
        && request.resource.data.nombre.size() > 0
        && request.resource.data.nombre.size() <= 60
        && request.resource.data.texto is string
        && request.resource.data.texto.size() > 0
        && request.resource.data.texto.size() <= 600
        && request.resource.data.ts is number;
      allow update, delete: if false;
    }

    match /canciones/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['tema','quien','ts'])
        && request.resource.data.tema is string
        && request.resource.data.tema.size() > 0
        && request.resource.data.tema.size() <= 120
        && request.resource.data.quien is string
        && request.resource.data.quien.size() <= 60
        && request.resource.data.ts is number;
      allow update, delete: if false;
    }

    match /confirmaciones/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['nombre','asiste','mensaje','ts'])
        && request.resource.data.nombre is string
        && request.resource.data.nombre.size() > 0
        && request.resource.data.nombre.size() <= 60
        && request.resource.data.asiste in ['si','no']
        && request.resource.data.mensaje is string
        && request.resource.data.mensaje.size() <= 300
        && request.resource.data.ts is number;
      allow update, delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
